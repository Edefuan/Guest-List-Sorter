function normalizePhone(raw) { const hadPlus = raw.trim().startsWith("+"); let digits = raw.replace(/\D/g, ""); // Some rows have a list number immediately before a Nigerian number: // for example, "1 080..." rather than just "080...". if (digits.length === 12 && /^[1-9]0/.test(digits)) { digits = digits.slice(1); } if (digits.startsWith("234") && digits.length === 13) { return "+" + digits; } // Nigerian local format: 08012345678 becomes +2348012345678. if (digits.length === 11 && digits.startsWith("0")) { return "+234" + digits.slice(1); } // Nigerian numbers sometimes appear without the first zero. if (digits.length === 10) { return "+234" + digits; } // Preserve numbers already written with an international country code. if (hadPlus && digits.length >= 8 && digits.length <= 15) { return "+" + digits; } // Leave unclear numbers for you to review instead of guessing. return raw.trim(); } function findRows(text) { const lines = text.split(/\r?\n/).map(line => line.trim()).filter(Boolean); const found = []; // Find phone-like number strings, including numbers separated by spaces or dashes. const phonePattern = /(?:\+\s*\d[\d().\s-]{8,}\d|\b\d[\d().\s-]{8,}\d\b)/g; for (let lineIndex = 0; lineIndex < lines.length; lineIndex++) { const line = lines[lineIndex]; const matches = [...line.matchAll(phonePattern)]; let previousEnd = 0; for (const match of matches) { let name = line.slice(previousEnd, match.index); // Remove common table headings and row numbers from the name. name = name .replace(/\b(?:Merlot|Violet)\s+table\s+\d+\b/ig, "") .replace(/\bName\s+Phone\s+number\b/ig, "") .replace(/[|,:;\t]+/g, " ") .replace(/\s+\d+\s*$/, "") .trim(); // Some Word tables put the name and phone in separate text lines. if (!name && lineIndex > 0) { const previousLine = lines[lineIndex - 1]; if (!/\btable\b|\bphone\b|\bname\b/i.test(previousLine) && !phonePattern.test(previousLine)) { name = previousLine.trim(); } phonePattern.lastIndex = 0; } found.push({ name, phone: normalizePhone(match[0]) }); previousEnd = match.index + match[0].length; } } return found; }
